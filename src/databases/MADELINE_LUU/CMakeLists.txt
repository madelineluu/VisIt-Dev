# CMakeLists.txt for MADELINE_LUU database plugin
# Follows the standard VisIt internal plugin pattern used by all src/databases plugins.

SET(COMMON_SOURCES
    MADELINE_LUUPluginInfo.C
    MADELINE_LUUCommonPluginInfo.C
    avt_MADELINE_LUUFileFormat.C
)

SET(LIBI_SOURCES
    MADELINE_LUUPluginInfo.C
)

SET(LIBM_SOURCES
    MADELINE_LUUMDServerPluginInfo.C
    ${COMMON_SOURCES}
)

SET(LIBE_SOURCES
    MADELINE_LUUEnginePluginInfo.C
    ${COMMON_SOURCES}
)

INCLUDE_DIRECTORIES(
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${VISIT_DATABASE_INCLUDES}
)

ADD_LIBRARY(IMDELINE_LUUDatabase ${LIBI_SOURCES})
TARGET_LINK_LIBRARIES(IMDELINE_LUUDatabase visitcommon)
SET(INSTALLTARGETS IMDELINE_LUUDatabase)

IF(NOT VISIT_ENGINE_ONLY AND NOT VISIT_DBIO_ONLY)
    ADD_LIBRARY(MMDELINE_LUUDatabase ${LIBM_SOURCES})
    TARGET_LINK_LIBRARIES(MMDELINE_LUUDatabase visitcommon avtdbatts avtdatabase_ser)
    ADD_TARGET_DEFINITIONS(MMDELINE_LUUDatabase SERVER)
    SET(INSTALLTARGETS ${INSTALLTARGETS} MMDELINE_LUUDatabase)
ENDIF(NOT VISIT_ENGINE_ONLY AND NOT VISIT_DBIO_ONLY)

ADD_LIBRARY(EMDELINE_LUUDatabase_ser ${LIBE_SOURCES})
TARGET_LINK_LIBRARIES(EMDELINE_LUUDatabase_ser visitcommon avtdbatts avtdatabase_ser)
ADD_TARGET_DEFINITIONS(EMDELINE_LUUDatabase_ser ENGINE)
SET(INSTALLTARGETS ${INSTALLTARGETS} EMDELINE_LUUDatabase_ser)

IF(VISIT_PARALLEL)
    ADD_PARALLEL_LIBRARY(EMDELINE_LUUDatabase_par ${LIBE_SOURCES})
    TARGET_LINK_LIBRARIES(EMDELINE_LUUDatabase_par visitcommon avtdbatts avtdatabase_par)
    ADD_TARGET_DEFINITIONS(EMDELINE_LUUDatabase_par ENGINE)
    SET(INSTALLTARGETS ${INSTALLTARGETS} EMDELINE_LUUDatabase_par)
ENDIF(VISIT_PARALLEL)

VISIT_INSTALL_DATABASE_PLUGINS(${INSTALLTARGETS})
